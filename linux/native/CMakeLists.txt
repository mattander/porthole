cmake_minimum_required(VERSION 3.21)
project(porthole VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets Network DBus Test)
find_package(LibIcal REQUIRED CONFIG)
add_library(porthole-typography typography.cpp fonts.qrc)
target_link_libraries(porthole-typography PUBLIC Qt6::Gui)
target_compile_options(porthole-typography PRIVATE -Wall -Wextra -Wpedantic)
add_library(porthole-profile profile.cpp)
target_link_libraries(porthole-profile PUBLIC Qt6::Network)
target_compile_options(porthole-profile PRIVATE -Wall -Wextra -Wpedantic)
add_library(porthole-data data.cpp diagnostic-log.cpp calendar-sources.cpp caldav.cpp briefing.cpp google-calendar.cpp)
target_link_libraries(porthole-data PUBLIC porthole-profile Qt6::Network ical)
target_compile_options(porthole-data PRIVATE -Wall -Wextra -Wpedantic)
add_library(porthole-dashboard dashboard-widgets.cpp event-widgets.cpp rain-forecast.cpp diagnostics.cpp calendar-style.cpp sun-theme.cpp display-controls.cpp graphics-diagnostics.cpp)
target_link_libraries(porthole-dashboard PUBLIC porthole-data porthole-typography Qt6::Widgets)
target_compile_options(porthole-dashboard PRIVATE -Wall -Wextra -Wpedantic)
add_executable(porthole main.cpp keyboard.cpp wifi.cpp settings.cpp calendar-client.cpp briefing-widget.cpp update-client.cpp)
target_link_libraries(porthole PRIVATE porthole-data porthole-dashboard Qt6::Widgets Qt6::DBus)
target_compile_options(porthole PRIVATE -Wall -Wextra -Wpedantic)
add_executable(porthole-calendar calendar-service.cpp)
target_link_libraries(porthole-calendar PRIVATE porthole-data)
target_compile_options(porthole-calendar PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(porthole-graphics-tests graphics-diagnostics-tests.cpp)
  target_link_libraries(porthole-graphics-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME graphics-diagnostics-and-legacy-export COMMAND porthole-graphics-tests)
  set_tests_properties(graphics-diagnostics-and-legacy-export PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

  add_executable(porthole-display-tests display-controls-tests.cpp)
  target_link_libraries(porthole-display-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME display-modes-sensor-and-backlight COMMAND porthole-display-tests)

  add_executable(porthole-agenda-tests agenda-tests.cpp calendar-client.cpp)
  target_link_libraries(porthole-agenda-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME agenda-details-navigation-and-hourly-rain COMMAND porthole-agenda-tests)
  set_tests_properties(agenda-details-navigation-and-hourly-rain PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(porthole-sun-theme-tests sun-theme-tests.cpp)
  target_link_libraries(porthole-sun-theme-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME sunrise-sunset-theme-and-persistence COMMAND porthole-sun-theme-tests)
  add_executable(porthole-google-tests google-tests.cpp)
  target_link_libraries(porthole-google-tests PRIVATE porthole-data Qt6::Test)
  add_test(NAME google-oauth-calendar-boundaries COMMAND porthole-google-tests)
  add_executable(porthole-profile-tests profile-tests.cpp)
  target_link_libraries(porthole-profile-tests PRIVATE porthole-data porthole-dashboard Qt6::Test)
  add_test(NAME private-profile-validation-and-persistence COMMAND porthole-profile-tests)
  add_executable(porthole-sync-tests sync-tests.cpp)
  target_link_libraries(porthole-sync-tests PRIVATE porthole-data Qt6::Test)
  add_test(NAME calendar-change-detection COMMAND porthole-sync-tests)
  add_executable(porthole-tests tests.cpp)
  target_link_libraries(porthole-tests PRIVATE porthole-data Qt6::Test)
  add_test(NAME calendar-and-auth-boundaries COMMAND porthole-tests)
  add_executable(porthole-ui-tests ui-tests.cpp keyboard.cpp wifi.cpp settings.cpp update-client.cpp)
  target_link_libraries(porthole-ui-tests PRIVATE porthole-dashboard Qt6::Widgets Qt6::DBus Qt6::Test)
  add_test(NAME settings-and-touch-entry COMMAND porthole-ui-tests)
  set_tests_properties(settings-and-touch-entry PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(porthole-wifi-tests wifi-tests.cpp wifi.cpp keyboard.cpp)
  target_link_libraries(porthole-wifi-tests PRIVATE porthole-typography Qt6::Widgets Qt6::DBus Qt6::Test)
  add_test(NAME wifi-ui-responsiveness COMMAND porthole-wifi-tests)
  set_tests_properties(wifi-ui-responsiveness PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(porthole-briefing-tests briefing-tests.cpp briefing-widget.cpp calendar-client.cpp)
  target_link_libraries(porthole-briefing-tests PRIVATE porthole-dashboard Qt6::Widgets Qt6::Test)
  add_test(NAME briefing-budget-privacy-and-layout COMMAND porthole-briefing-tests)
  set_tests_properties(briefing-budget-privacy-and-layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(porthole-colour-tests colour-tests.cpp briefing-widget.cpp)
  target_link_libraries(porthole-colour-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME calendar-colours-and-briefing-tints COMMAND porthole-colour-tests)
  set_tests_properties(calendar-colours-and-briefing-tints PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(porthole-layout-tests layout-tests.cpp)
  target_link_libraries(porthole-layout-tests PRIVATE porthole-dashboard Qt6::Test)
  add_test(NAME dashboard-clock-footer-and-diagnostics COMMAND porthole-layout-tests)
  set_tests_properties(dashboard-clock-footer-and-diagnostics PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(porthole-update-tests update-client-tests.cpp update-client.cpp)
  target_link_libraries(porthole-update-tests PRIVATE Qt6::Network Qt6::Test)
  add_test(NAME software-update-ipc-boundaries COMMAND porthole-update-tests)
endif()
install(TARGETS porthole porthole-calendar RUNTIME DESTINATION bin)
